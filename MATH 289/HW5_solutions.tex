\section*{Problem 1}

Let
\[
J_m(\lambda)=
\begin{pmatrix}
\lambda&1&0&\cdots&0\\
0&\lambda&1&\ddots&\vdots\\
\vdots&\ddots&\ddots&\ddots&0\\
0&\cdots&0&\lambda&1\\
0&\cdots&\cdots&0&\lambda
\end{pmatrix}.
\]
Write $J_m(\lambda)^n=(c^{(n)}_{ij})$. Since the only nonzero entries
in column $j$ of $J_m(\lambda)$ are $\lambda$ in row $j$ and $1$ in
row $j-1$ (when $j>1$), matrix multiplication gives
\[
c^{(n+1)}_{ij}=\lambda c^{(n)}_{ij}+c^{(n)}_{i,j-1},
\]
where an index outside $1,\ldots,m$ contributes $0$.

We claim that, for every $n\geq0$,
\[
c^{(n)}_{ij}=\begin{cases}
\displaystyle\binom{n}{j-i}\lambda^{\,n-(j-i)},
&0\leq j-i\leq\min\{n,m-1\},\\[0.7em]
0,&\text{otherwise.}
\end{cases}
\]
For $n=0$, this is the identity matrix. Assume it holds for $n$. If
$k=j-i$ satisfies $1\leq k\leq n$, then the recurrence and Pascal's
identity give
\[
c^{(n+1)}_{i,i+k}
=\lambda\binom{n}{k}\lambda^{n-k}
 +\binom{n}{k-1}\lambda^{n-k+1}
=\binom{n+1}{k}\lambda^{n+1-k}.
\]
For $k=0$ the recurrence gives $\lambda^{n+1}$; for $k=n+1$ the
second term gives $1$; and for $k>n+1$ both terms are zero. Thus the
claim holds by induction.

Therefore
\[
\boxed{
\bigl(J_m(\lambda)^n\bigr)_{ij}
=\begin{cases}
\displaystyle\binom{n}{j-i}\lambda^{\,n-(j-i)},
&0\leq j-i\leq\min\{n,m-1\},\\[0.7em]
0,&\text{otherwise.}
\end{cases}}
\]
If $b_k=\binom{n}{k}\lambda^{n-k}$ for $0\leq k\leq n$ and $b_k=0$
for $k>n$, this is
\[
\boxed{
J_m(\lambda)^n=
\begin{pmatrix}
b_0&b_1&b_2&\cdots&b_{m-1}\\
0&b_0&b_1&\ddots&\vdots\\
\vdots&\ddots&\ddots&\ddots&b_2\\
0&\cdots&0&b_0&b_1\\
0&\cdots&\cdots&0&b_0
\end{pmatrix}.}
\]

\section*{Problem 2}

Let $A=(a_{ij})_{i,j=1}^n$, where $a_{ii}=2$ and
$a_{ij}=(-1)^{|i-j|}$ for $i\ne j$. Multiply row $i$ and column $i$
by $(-1)^{i-1}$ for every $i$. Equivalently, first multiply each row
by these signs and then multiply each column by the same signs. The
determinant is unchanged because the two products of signs square to
$1$.

Call the resulting matrix $B$. Its diagonal entries are $2$. For
$i\ne j$,
\[
B_{ij}=(-1)^{i-1}(-1)^{|i-j|}(-1)^{j-1}=1,
\]
because $|i-j|$, $i-j$, and $i+j$ have the same parity. Thus $B$ has
2 on the diagonal and 1 everywhere else.

For $i=2,\ldots,n$, replace row $i$ by row $i$ minus row $1$. The
first row remains $(2,1,\ldots,1)$, and each later row becomes
\[
(-1,0,\ldots,0,1,0,\ldots,0),
\]
with the $1$ in column $i$. Now replace column $1$ by the sum of all
columns. The resulting matrix is upper triangular: its first diagonal
entry is
\[
2-(n-1)(-1)=n+1,
\]
and every other diagonal entry is $1$. These row and column operations
preserve the determinant, so
\[
\boxed{\det A=\det B=(n+1)\cdot1^{n-1}=n+1.}
\]
